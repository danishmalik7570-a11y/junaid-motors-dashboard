class Car(models.Model):
    STATUS_CHOICES = [
        ('available', 'Available'),
        ('reserved', 'Reserved'),
        ('sold', 'Sold'),
        ('under_repair', 'Under Repair'),
    ]
    name = models.CharField(max_length=200)
    model_year = models.IntegerField()
    color = models.CharField(max_length=100)
    registration_no = models.CharField(max_length=50, unique=True)
    purchase_price = models.DecimalField(max_digits=12, decimal_places=2)
    selling_price = models.DecimalField(max_digits=12, decimal_places=2)
    status = models.CharField(max_length=20, choices=STATUS_CHOICES, default='available')
    image = models.ImageField(upload_to='cars/', blank=True, null=True)
    entry_date = models.DateField(auto_now_add=True)
    notes = models.TextField(blank=True)

    def profit_margin(self):
        return self.selling_price - self.purchase_price

    def days_in_stock(self):
        from django.utils import timezone
        return (timezone.now().date() - self.entry_date).days

    def __str__(self):
        return f"{self.name} ({self.model_year}) - {self.registration_no}"

class Sale(models.Model):
    PAYMENT_CHOICES = [
        ('cash', 'Cash'),
        ('installment', 'Installment'),
    ]
    invoice_no = models.CharField(max_length=20, unique=True)
    car = models.ForeignKey('inventory.Car', on_delete=models.PROTECT)
    customer = models.ForeignKey('customers.Customer', on_delete=models.PROTECT)
    sale_date = models.DateField(auto_now_add=True)
    payment_type = models.CharField(max_length=20, choices=PAYMENT_CHOICES)
    total_amount = models.DecimalField(max_digits=12, decimal_places=2)
    down_payment = models.DecimalField(max_digits=12, decimal_places=2, default=0)
    monthly_installment = models.DecimalField(max_digits=10, decimal_places=2, default=0)
    installment_months = models.IntegerField(default=0)
    profit = models.DecimalField(max_digits=12, decimal_places=2)

    def save(self, *args, **kwargs):
        if not self.invoice_no:
            last = Sale.objects.order_by('-id').first()
            num = (last.id + 1) if last else 1
            self.invoice_no = f"INV-{num:05d}"
        self.profit = self.total_amount - self.car.purchase_price
        super().save(*args, **kwargs)
        self.car.status = 'sold'
        self.car.save()

    def __str__(self):
        return f"{self.invoice_no} - {self.car.name} to {self.customer.name}"

class Installment(models.Model):
    STATUS_CHOICES = [
        ('pending', 'Pending'),
        ('paid', 'Paid'),
        ('overdue', 'Overdue'),
    ]
    sale = models.ForeignKey('sales.Sale', on_delete=models.CASCADE)
    customer = models.ForeignKey('customers.Customer', on_delete=models.CASCADE)
    installment_no = models.IntegerField()
    due_date = models.DateField()
    amount = models.DecimalField(max_digits=10, decimal_places=2)
    status = models.CharField(max_length=10, choices=STATUS_CHOICES, default='pending')
    paid_date = models.DateField(blank=True, null=True)
    late_fee = models.DecimalField(max_digits=8, decimal_places=2, default=0)

    def check_overdue(self):
        from django.utils import timezone
        if self.status == 'pending' and self.due_date < timezone.now().date():
            self.status = 'overdue'
            self.save()

    def __str__(self):
        return f"Installment #{self.installment_no} - {self.customer.name}"

@login_required
def dashboard(request):
    today = timezone.now().date()
    this_month_start = today.replace(day=1)

    context = {
        'total_cars_available': Car.objects.filter(status='available').count(),
        'sold_this_month': Sale.objects.filter(sale_date__gte=this_month_start).count(),
        'pending_installments': Installment.objects.filter(status__in=['pending','overdue']).count(),
        'overdue_installments': Installment.objects.filter(status='overdue').count(),
        'monthly_revenue': Sale.objects.filter(sale_date__gte=this_month_start).aggregate(
            total=Sum('total_amount'))['total'] or 0,
        'monthly_profit': Sale.objects.filter(sale_date__gte=this_month_start).aggregate(
            total=Sum('profit'))['total'] or 0,
        'new_leads': Lead.objects.filter(status='new').count(),
        'total_customers': Customer.objects.count(),
        'recent_sales': Sale.objects.select_related('car','customer').order_by('-sale_date')[:5],
        'low_stock_warning': Car.objects.filter(status='available').count() < 5,
        'today_sales': Sale.objects.filter(sale_date=today),
        'slow_moving_cars': Car.objects.filter(
            status='available',
            entry_date__lte=today - timedelta(days=30)
        ).count(),
        'monthly_sales_chart': get_monthly_sales_data(),
    }
    return render(request, 'dashboard/dashboard.html', context)

def get_monthly_sales_data():
    data = []
    today = timezone.now().date()
    for i in range(5, -1, -1):
        month = today.replace(day=1) - timedelta(days=i*28)
        count = Sale.objects.filter(
            sale_date__year=month.year,
            sale_date__month=month.month
        ).count()
        revenue = Sale.objects.filter(
            sale_date__year=month.year,
            sale_date__month=month.month
        ).aggregate(total=Sum('total_amount'))['total'] or 0
        data.append({
            'month': month.strftime('%b %Y'),
            'count': count,
            'revenue': float(revenue)
        })
    return data

@login_required
def inventory_list(request):
    cars = Car.objects.all().order_by('-entry_date')
    status_filter = request.GET.get('status', '')
    search = request.GET.get('search', '')
    if status_filter:
        cars = cars.filter(status=status_filter)
    if search:
        cars = cars.filter(
            Q(name__icontains=search) |
            Q(registration_no__icontains=search) |
            Q(color__icontains=search)
        )
    return render(request, 'inventory/list.html', {'cars': cars, 'status_filter': status_filter, 'search': search})

@login_required
def new_sale(request):
    if request.method == 'POST':
        form = SaleForm(request.POST)
        if form.is_valid():
            sale = form.save(commit=False)
            sale.profit = sale.total_amount - sale.car.purchase_price
            sale.save()
            # Auto-generate installments if payment type is installment
            if sale.payment_type == 'installment' and sale.installment_months > 0:
                for i in range(1, sale.installment_months + 1):
                    due = sale.sale_date + timedelta(days=30 * i)
                    Installment.objects.create(
                        sale=sale,
                        customer=sale.customer,
                        installment_no=i,
                        due_date=due,
                        amount=sale.monthly_installment,
                        status='pending'
                    )
            messages.success(request, f'Sale recorded. Invoice: {sale.invoice_no}')
            return redirect('sale_detail', pk=sale.pk)
    else:
        form = SaleForm()
    available_cars = Car.objects.filter(status='available')
    customers = Customer.objects.all()
    return render(request, 'sales/new_sale.html', {
        'form': form,
        'available_cars': available_cars,
        'customers': customers
    })

@login_required
def installment_list(request):
    today = timezone.now().date()
    # Auto-update overdue status
    Installment.objects.filter(status='pending', due_date__lt=today).update(status='overdue')

    status_filter = request.GET.get('status', '')
    installments = Installment.objects.select_related(
        'sale__car', 'customer'
    ).order_by('due_date')
    if status_filter:
        installments = installments.filter(status=status_filter)

    summary = {
        'total_expected': installments.filter(
            due_date__month=today.month).aggregate(Sum('amount'))['amount__sum'] or 0,
        'collected': installments.filter(
            status='paid', paid_date__month=today.month).aggregate(Sum('amount'))['amount__sum'] or 0,
        'overdue_amount': installments.filter(
            status='overdue').aggregate(Sum('amount'))['amount__sum'] or 0,
        'pending_count': installments.filter(status__in=['pending','overdue']).count(),
    }
    return render(request, 'installments/list.html', {
        'installments': installments, 'summary': summary, 'status_filter': status_filter
    })

@login_required
def pnl_report(request):
    sales = Sale.objects.all()
    total_revenue = sales.aggregate(Sum('total_amount'))['total_amount__sum'] or 0
    total_profit = sales.aggregate(Sum('profit'))['profit__sum'] or 0
    total_cost = Purchase.objects.aggregate(
        total=Sum(F('purchase_price') + F('dealer_commission')))['total'] or 0
    total_commission = Purchase.objects.aggregate(Sum('dealer_commission'))['dealer_commission__sum'] or 0

    monthly_pnl = []
    today = timezone.now().date()
    for i in range(5, -1, -1):
        month = today.replace(day=1) - timedelta(days=i*28)
        month_sales = Sale.objects.filter(
            sale_date__year=month.year, sale_date__month=month.month)
        monthly_pnl.append({
            'month': month.strftime('%B %Y'),
            'cars_sold': month_sales.count(),
            'revenue': month_sales.aggregate(Sum('total_amount'))['total_amount__sum'] or 0,
            'profit': month_sales.aggregate(Sum('profit'))['profit__sum'] or 0,
        })

    context = {
        'total_revenue': total_revenue,
        'total_cost': total_cost,
        'total_profit': total_profit,
        'total_commission': total_commission,
        'profit_margin': round((total_profit / total_revenue * 100), 2) if total_revenue > 0 else 0,
        'monthly_pnl': monthly_pnl,
        'best_selling': Car.objects.filter(status='sold').annotate(
            sale_count=Count('sale')).order_by('-sale_count')[:5],
        'slow_moving': Car.objects.filter(
            status='available',
            entry_date__lte=today - timedelta(days=30)
        ).order_by('entry_date'),
    }
    return render(request, 'reports/pnl.html', context)

class RoleRequiredMixin:

    def dispatch(self, request, *args, **kwargs):
        if hasattr(request.user, 'staffprofile'):
            if request.user.staffprofile.role in self.required_roles or 'admin' in self.required_roles:
                return super().dispatch(request, *args, **kwargs)
        return redirect('dashboard')
